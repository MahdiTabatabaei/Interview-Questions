تعریف Minimal APIها در  ASP.NET Core (معرفی شده در .NET 6) روشی ساده‌تر و مختصرتر برای ساخت Endpointهای HTTP بدون نیاز به ایجاد Controllerها و Viewهای کامل MVC یا Razor Pages هستند. این APIها برای ساخت برنامه‌های کوچک، میکروسرویس‌ها، یا Endpointهای ساده‌ای که نیاز به منطق پیچیده یا UI ندارند، ایده‌آل هستند.

**ویژگی‌ها:**

- **کد کمتر:** نیازی به کلاس‌های Controller، Action Methodها، یا فایل‌های View نیست.
- **استفاده مستقیم از `Program.cs`:** تعریف Endpointها مستقیماً در فایل `Program.cs` انجام می‌شود.
- **سادگی:** برای کارهای ساده بسیار سریع و آسان هستند.
- **پشتیبانی از DI:** همچنان از سیستم Dependency Injection برای تزریق سرویس‌ها پشتیبانی می‌کنند.

**چگونه کار می‌کند:**

از متدهایی مانند `MapGet`, `MapPost`, `MapPut`, `MapDelete` که روی شیء `WebApplication` (یا `IEndpointRouteBuilder`) فراخوانی می‌شوند، برای تعریف Endpointها استفاده می‌شود. این متدها یک الگو (Route Pattern) و یک delegate (معمولاً با لامبدا اکسپرشن) که کد پردازش درخواست را اجرا می‌کند، می‌گیرند.

**مثال:**
```csharp

// Program.cs
var builder = WebApplication.CreateBuilder(args);

// Register services if needed (e.g., DbContext, custom services)
builder.Services.AddScoped<IProductService, ProductService>();

var app = builder.Build();

// Configure the HTTP request pipeline.
if (app.Environment.IsDevelopment())
{
    app.UseDeveloperExceptionPage();
}

app.UseRouting(); // Essential for attribute routing and conventional routing, but Map* methods include endpoint routing.

// Define Minimal API Endpoints
app.MapGet("/", () => "Hello World!"); // یک Endpoint ساده

app.MapGet("/products", async (IProductService productService) =>
{
    var products = await productService.GetAllAsync();
    return Results.Ok(products); // بازگرداندن JSON با Status Code 200 OK
});

app.MapGet("/products/{id}", async (int id, IProductService productService) =>
{
    var product = await productService.GetByIdAsync(id);
    if (product == null)
    {
        return Results.NotFound($"Product with ID {id} not found."); // بازگرداندن 404 Not Found
    }
    return Results.Ok(product); // بازگرداندن JSON با Status Code 200 OK
});

app.MapPost("/products", async (Product newProduct, IProductService productService) =>
{
    var createdProduct = await productService.AddAsync(newProduct);
    // بازگرداندن 201 Created با URL محصول جدید و خود محصول
    return Results.Created($"/products/{createdProduct.Id}", createdProduct);
});

app.Run();

// ---- Supporting Classes ----
public class Product
{
    public int Id { get; set; }
    public string Name { get; set; }
    public decimal Price { get; set; }
}

public interface IProductService
{
    Task<IEnumerable<Product>> GetAllAsync();
    Task<Product> GetByIdAsync(int id);
    Task<Product> AddAsync(Product product);
}

public class ProductService : IProductService
{
    // Mock data for demonstration
    private static List<Product> _products = new List<Product>
    {
        new Product { Id = 1, Name = "Laptop", Price = 1200 },
        new Product { Id = 2, Name = "Mouse", Price = 25 }
    };
    private static int _nextId = 3;

    public Task<IEnumerable<Product>> GetAllAsync() => Task.FromResult(_products.AsEnumerable());

    public Task<Product> GetByIdAsync(int id) => Task.FromResult(_products.FirstOrDefault(p => p.Id == id));

    public Task<Product> AddAsync(Product product)
    {
        product.Id = _nextId++;
        _products.Add(product);
        return Task.FromResult(product);
    }
}

```
